function Start-AffServer {
    param([int]$Nc,[int]$Cache,[int]$Threads,[IntPtr]$Mask,[string]$Tag)
    $sargs = @('-m',$Model,'-ngl','999','-ncmoe',"$Nc",'--moe-cache-mib',"$Cache",
               '--host','127.0.0.1','--port',"$Port",'-c','8192','-b','2048','-ub','512',
               '-np','1','-t',"$Threads",'-ctk','f16','-ctv','f16','-fa','on')
    $err = Join-Path $Out "srv_$Tag.err"
    $p = Start-Process -FilePath $Server -ArgumentList $sargs -PassThru -WindowStyle Hidden `
         -RedirectStandardError $err -RedirectStandardOutput (Join-Path $Out "srv_$Tag.out")
    $ready = $false
    for ($i=0; $i -lt 240; $i++) {
        Start-Sleep -Milliseconds 1000
        try { $h = Invoke-RestMethod -Uri "http://127.0.0.1:$Port/health" -TimeoutSec 3
              if ($h.status -eq 'ok') { $ready = $true; break } } catch {}
    }
    if (-not $ready) { Log "NOT READY $Tag"; return $null }
    # apply affinity AFTER the process exists and has spawned its threads
    try { $p.ProcessorAffinity = $Mask; Log "affinity $Tag set to 0x$('{0:X}' -f [int64]$Mask)" }
    catch { Log "AFFINITY SET FAILED $Tag : $($_.Exception.Message)" }
    Start-Sleep -Seconds 2
    Log "server $Tag ready in ${i}s threads=$Threads vram_free=$((Get-VramFree))"
    return [pscustomobject]@{ Proc=$p; Err=$err }
}
